GROQ SDK CONTROLLER (api/chat.js) 
import Groq from 'groq-sdk';

const groq = new Groq({ apiKey: process.env.GROQ_API_KEY });

export default async function handler(req, res) {
  try {
    const { crop, location, soil, irrigation, problem } = req.body;

    const systemPrompt = `आप 'AI किसान सहायक' हैं — भारतीय कृषि अनुसंधान परिषद (ICAR) व KVK मानकों पर आधारित विशेषज्ञ।
    सरल, व्यावहारिक व प्रमाणिक हिंदी में 5 भागों में जवाब दें:
    1. प्राथमिक निदान व कीट-रोग प्रबंधन (जैविक + रासायनिक मात्रा सहित)
    2. संतुलित खाद व उर्वरक खुराक (प्रति एकड़ यूरिया/डीएपी/जिंक)
    3. सटीक सिंचाई अवस्था
    4. कम लागत कृषि व्यवसाय विचार (मशरूम, वर्मीकम्पोस्ट आदि)
    5. संबंधित सरकारी योजना (PM-KISAN, KCC आदि)`;

    const completion = await groq.chat.completions.create({
      model: process.env.GROQ_MODEL || 'openai/gpt-oss-120b',
      messages: [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: `फसल: ${crop}, स्थान: ${location}, मिट्टी: ${soil}, सिंचाई: ${irrigation}, प्रश्न: ${problem}` }
      ],
      temperature: 0.2,
      max_tokens: 1500
    });

    res.json({ response: completion.choices[0].message.content });
  } catch (error) {
    console.error('Groq LPU Error:', error);
    res.status(500).json({ error: 'कृषि सलाहकार सर्वर प्रतिक्रिया में त्रुटि' });
  }
}
